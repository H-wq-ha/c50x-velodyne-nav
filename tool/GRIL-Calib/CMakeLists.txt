cmake_minimum_required(VERSION 3.16)
project(gril_calib)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_definitions(ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/" DEPLOY MP_PROC_NUM=1)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree segmentation sample_consensus)
find_package(Ceres REQUIRED)
find_package(OpenMP REQUIRED)
option(GRIL_WITH_LIVOX "Build optional Livox input" OFF)
if(GRIL_WITH_LIVOX)
  find_package(livox_ros_driver2 REQUIRED)
  add_compile_definitions(GRIL_WITH_LIVOX)
endif()
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Pose6D.msg")
ament_export_dependencies(rosidl_default_runtime)
include_directories(include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${CERES_INCLUDE_DIRS})
add_library(gril_floor STATIC src/automatic_floor.cpp)
target_link_libraries(gril_floor ${PCL_LIBRARIES})
add_executable(gril_calib_exec src/laserMapping.cpp src/preprocess.cpp
  include/ikd-Tree/ikd_Tree.cpp include/Gril_Calib/Gril_Calib.cpp
  include/Fusion/FusionAhrs.c include/Fusion/FusionCompass.c include/Fusion/FusionOffset.c)
ament_target_dependencies(gril_calib_exec rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs visualization_msgs tf2 tf2_ros pcl_conversions)
if(GRIL_WITH_LIVOX)
  ament_target_dependencies(gril_calib_exec livox_ros_driver2)
endif()
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(gril_calib_exec gril_floor ${PCL_LIBRARIES} ${CERES_LIBRARIES} OpenMP::OpenMP_CXX OpenMP::OpenMP_C "${cpp_typesupport_target}")
include(CTest)
if(BUILD_TESTING)
  add_executable(test_velodyne_time tests/test_velodyne_time.cpp src/preprocess.cpp)
  ament_target_dependencies(test_velodyne_time rclcpp sensor_msgs nav_msgs tf2_ros pcl_conversions)
  target_link_libraries(test_velodyne_time ${PCL_LIBRARIES} OpenMP::OpenMP_CXX "${cpp_typesupport_target}")
  add_test(NAME velodyne_point_times COMMAND test_velodyne_time)
  add_executable(test_planar_solver tests/test_planar_solver.cpp include/Gril_Calib/Gril_Calib.cpp)
  ament_target_dependencies(test_planar_solver rclcpp sensor_msgs nav_msgs tf2_ros pcl_conversions)
  target_link_libraries(test_planar_solver ${PCL_LIBRARIES} ${CERES_LIBRARIES} "${cpp_typesupport_target}")
  add_test(NAME synthetic_planar_calibration COMMAND test_planar_solver)
  set_tests_properties(synthetic_planar_calibration PROPERTIES TIMEOUT 120)
  add_executable(test_automatic_floor tests/test_automatic_floor.cpp)
  target_link_libraries(test_automatic_floor gril_floor ${PCL_LIBRARIES})
  add_test(NAME automatic_floor_without_height_prior COMMAND test_automatic_floor)
endif()
install(TARGETS gril_calib_exec DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch rviz_cfg DESTINATION share/${PROJECT_NAME})
ament_package()
